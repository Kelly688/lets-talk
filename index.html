        // Voting Guides Category Filtering

        function filterVoting(category) {

            const cards = document.querySelectorAll('.voting-card');

            cards.forEach(card => {

                if (category === 'all' || card.dataset.category === category) {

                    card.style.display = 'block';

                } else {

                    card.style.display = 'none';

                }

            });


            document.querySelectorAll('.voting-filter-btn').forEach(btn => {

                btn.className = "voting-filter-btn px-5 py-2 rounded-xl text-xs font-bold glass-card text-gray-700 transition-all";

            });


            const activeBtn = document.getElementById(`vbtn-${category}`);

            if (activeBtn) {

                activeBtn.className = "voting-filter-btn px-5 py-2 rounded-xl text-xs font-bold bg-gray-900 text-white shadow-md transition-all";

            }

        }


        // Bill Drafting Assistant Renderer

        function renderBillOutline() {

            const title = document.getElementById('billTitleInput').value.trim() || "[Title]";

            const jurisdiction = document.getElementById('billJurisdiction').value;

            const mandate = document.getElementById('billMandateInput').value.trim() || "[Mandate Description]";


            const template = `AN ACT CONCERNING: ${title.toUpperCase()}

Be it enacted by the ${jurisdiction}:


SECTION 1. SHORT TITLE.

This Act may be cited as the "${title}".


SECTION 2. PURPOSE AND LEGISLATIVE FINDINGS.

The legislative body finds that ${mandate} is in the vital interest of public health, safety, and economic welfare.


SECTION 3. PROVISIONS & IMPLEMENTATION.

(a) The designated regulatory authority shall promulgate rules to enforce: ${mandate}.

(b) Penalties for non-compliance shall be assessed pursuant to state statutory guidelines.


SECTION 4. EFFECTIVE DATE.

This Act shall take effect 90 days following formal enactment.`;


            document.getElementById('billPreviewText').innerText = template;

        }


        function copyBillOutline() {

            const text = document.getElementById('billPreviewText').innerText;

            navigator.clipboard.writeText(text);

            alert("Bill draft template copied to clipboard!");

        }


        // Media Tabs Switcher (Books / Podcasts)

        function switchMediaTab(tab) {

            const booksContent = document.getElementById('booksContent');

            const podcastsContent = document.getElementById('podcastsContent');

            const tabBooks = document.getElementById('tab-books');

            const tabPodcasts = document.getElementById('tab-podcasts');


            if (tab === 'books') {

                booksContent.classList.remove('hidden');

                podcastsContent.classList.add('hidden');

                tabBooks.className = "px-6 py-2.5 rounded-2xl text-xs font-bold bg-brandDark text-white shadow-md transition-all flex items-center gap-2";

                tabPodcasts.className = "px-6 py-2.5 rounded-2xl text-xs font-bold glass-card text-gray-700 hover:text-brandPurple transition-all flex items-center gap-2";

            } else {

                booksContent.classList.add('hidden');

                podcastsContent.classList.remove('hidden');

                tabPodcasts.className = "px-6 py-2.5 rounded-2xl text-xs font-bold bg-brandDark text-white shadow-md transition-all flex items-center gap-2";

                tabBooks.className = "px-6 py-2.5 rounded-2xl text-xs font-bold glass-card text-gray-700 hover:text-brandPurple transition-all flex items-center gap-2";

            }

        }


        // Search Legal Codes

        function searchLegalCodes() {

            const query = document.getElementById('legalSearchInput').value.toLowerCase();

            const items = document.querySelectorAll('.legal-item');


            items.forEach(item => {

                const text = item.innerText.toLowerCase();

                if (text.includes(query)) {

                    item.style.display = 'block';

                } else {

                    item.style.display = 'none';

                }

            });

        }
